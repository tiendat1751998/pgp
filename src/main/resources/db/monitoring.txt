# ============================================================
# DATABASE MONITORING CONFIGURATION
# Prometheus + Grafana Metrics
# ============================================================

# Datasource Metrics (HikariCP)
spring.datasource.hikari.metrics.enabled=true
spring.datasource.hikari.metrics.metric-names=hikari

# Additional Metrics
management.metrics.enable.hikaricp=true
management.metrics.enable.jpa=true
management.metrics.enable.hibernate=true

# ============================================================
# Key Metrics to Monitor:
# ============================================================

# 1. HikariCP Pool
# - hikaricp_connections_active
# - hikaricp_connections_idle
# - hikaricp_connections_pending
# - hikaricp_connections_max
# - hikaricp_connections_min
# - hikaricp_connections_acquire_seconds
# - hikaricp_connections_creation_seconds
# - hikaricp_connections_timeout_total_seconds

# 2. Query Performance
# - hibernate_seconds_total (query execution time)
# - hibernate_entities_insert_total
# - hibernate_entities_fetch_total
# - hibernate_queries_executed_total

# 3. Database Connection
# - db_connections_active
# - db_connections_idle
# - db_wait_count_total
